    Add a module-level helper function (after the `_diag_log` function, before the class):
    ```python
    def apply_curve_client(socket, server_public_key_z85: str):
        """Apply CURVE client encryption to a ZMQ socket.
        MUST be called BEFORE socket.connect().
        Args:
            socket: ZMQ socket (REQ, SUB, etc.)
            server_public_key_z85: Server's Z85-encoded public key (40-char string)
        """
        client_public, client_secret = zmq.curve_keypair()
        socket.curve_publickey = client_public
        socket.curve_secretkey = client_secret
        socket.curve_serverkey = server_public_key_z85.encode('ascii')
        logger.info("CURVE client configured (server key: %s...)", server_public_key_z85[:10])
    ```

        self.curve_enabled = curve_enabled

    Modify `ZMQClient.connect()` to apply CURVE BEFORE socket.connect():
    ```python
    def connect(self) -> bool:
        try:
            self.context = zmq.Context()
            self.socket = self.context.socket(zmq.REQ)
            self.socket.setsockopt(zmq.RCVTIMEO, self.timeout)

            # Apply CURVE encryption BEFORE connect (ordering is critical)
            if self.curve_enabled and self.server_public_key:
                apply_curve_client(self.socket, self.server_public_key)
                print(f"[ZMQ] CURVE enabled (server key: {self.server_public_key[:10]}...)")
            else:
                print(f"[ZMQ] CURVE disabled (plaintext mode)")

            self.socket.connect(f"tcp://{self.host}:{self.port}")
            print(f"[ZMQ] SUCCESS: Connected to tcp://{self.host}:{self.port}")
            return True
        except Exception as e:
            print(f"[ZMQ] ERROR: Connection failed: {e}")
            return False
    ```
